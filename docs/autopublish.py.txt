#!/usr/bin/env python3
"""
Автопостинг шортсов на YouTube через YouTube Data API v3.

Файлы рядом со скриптом:
  client_secret.json — OAuth-клиент (Desktop) из Google Cloud, НЕ публиковать
  token.json         — создаётся после входа, НЕ публиковать

Команды:
  python3 autopublish.py auth
      войти в аккаунт канала (откроется браузер) и показать, какой канал подключён
  python3 autopublish.py upload VIDEO.mp4 --title "..." [--desc "..."] [--tags a,b] [--at "2026-11-21 18:00"]
      загрузить; --at — время публикации (по часовому поясу этого Mac), без --at — сразу публично
  python3 autopublish.py queue queue.json
      загрузить всё из очереди: [{"file": "...", "title": "...", "desc": "...", "tags": [...], "at": "..."}]
      уже загруженные помечаются "video_id" и пропускаются

ВАЖНО: пока Google не одобрил проект (audit YouTube API), загруженные через API видео
YouTube принудительно делает приватными — отложенная публикация сработает только после одобрения.
"""
import argparse
import json
import sys
from datetime import datetime, timezone
from pathlib import Path

from google.auth.transport.requests import Request
from google.oauth2.credentials import Credentials
from google_auth_oauthlib.flow import InstalledAppFlow
from googleapiclient.discovery import build
from googleapiclient.http import MediaFileUpload

HERE = Path(__file__).resolve().parent
SECRET = HERE / "client_secret.json"
TOKEN = HERE / "token.json"
SCOPES = ["https://www.googleapis.com/auth/youtube.upload",
          "https://www.googleapis.com/auth/youtube.readonly"]
CATEGORY_GAMING = "20"


def service():
    creds = None
    if TOKEN.exists():
        creds = Credentials.from_authorized_user_file(str(TOKEN), SCOPES)
    if not creds or not creds.valid:
        if creds and creds.expired and creds.refresh_token:
            try:
                creds.refresh(Request())
            except Exception:
                creds = None  # в режиме тестирования токен живёт 7 дней — входим заново
        if not creds or not creds.valid:
            if not SECRET.exists():
                sys.exit(f"Нет {SECRET} — положи сюда OAuth-клиент из Google Cloud")
            flow = InstalledAppFlow.from_client_secrets_file(str(SECRET), SCOPES)
            creds = flow.run_local_server(port=0, prompt="consent")
        TOKEN.write_text(creds.to_json())
        TOKEN.chmod(0o600)
    return build("youtube", "v3", credentials=creds)


def to_utc_iso(local: str) -> str:
    dt = datetime.strptime(local, "%Y-%m-%d %H:%M").astimezone()  # локальное время Mac
    if dt <= datetime.now().astimezone():
        sys.exit(f"Время публикации {local} уже прошло")
    return dt.astimezone(timezone.utc).strftime("%Y-%m-%dT%H:%M:%S.000Z")


def upload(yt, file, title, desc="", tags=None, at=None):
    path = Path(file).expanduser()
    if not path.exists():
        raise FileNotFoundError(path)
    status = {"selfDeclaredMadeForKids": False}
    if at:
        status.update(privacyStatus="private", publishAt=to_utc_iso(at))
    else:
        status["privacyStatus"] = "public"
    body = {
        "snippet": {"title": title[:100], "description": desc, "tags": tags or [],
                    "categoryId": CATEGORY_GAMING},
        "status": status,
    }
    req = yt.videos().insert(part="snippet,status", body=body,
                             media_body=MediaFileUpload(str(path), chunksize=-1, resumable=True))
    resp = None
    while resp is None:
        _, resp = req.next_chunk()
    vid = resp["id"]
    st = resp.get("status", {})
    print(f"✓ {path.name} -> https://youtube.com/shorts/{vid}  "
          f"(status: {st.get('privacyStatus')}, publishAt: {st.get('publishAt', '—')})")
    return vid


def main():
    ap = argparse.ArgumentParser(description="Автопостинг на YouTube")
    sub = ap.add_subparsers(dest="cmd", required=True)
    sub.add_parser("auth")
    u = sub.add_parser("upload")
    u.add_argument("file")
    u.add_argument("--title", required=True)
    u.add_argument("--desc", default="")
    u.add_argument("--tags", default="")
    u.add_argument("--at", help='"YYYY-MM-DD HH:MM" по времени этого Mac')
    q = sub.add_parser("queue")
    q.add_argument("queue_file")
    a = ap.parse_args()

    yt = service()
    if a.cmd == "auth":
        ch = yt.channels().list(part="snippet,statistics", mine=True).execute().get("items", [])
        if not ch:
            sys.exit("Вход выполнен, но у аккаунта нет YouTube-канала")
        c = ch[0]
        print(f"Подключён канал: {c['snippet']['title']}  "
              f"(подписчиков: {c['statistics'].get('subscriberCount', '?')}, "
              f"видео: {c['statistics'].get('videoCount', '?')})")
    elif a.cmd == "upload":
        upload(yt, a.file, a.title, a.desc, [t.strip() for t in a.tags.split(",") if t.strip()], a.at)
    elif a.cmd == "queue":
        qf = Path(a.queue_file)
        items = json.loads(qf.read_text(encoding="utf-8"))
        for it in items:
            if it.get("video_id"):
                continue
            it["video_id"] = upload(yt, it["file"], it["title"], it.get("desc", ""), it.get("tags"), it.get("at"))
            qf.write_text(json.dumps(items, ensure_ascii=False, indent=2), encoding="utf-8")


if __name__ == "__main__":
    main()
